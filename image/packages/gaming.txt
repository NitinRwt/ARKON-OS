# ARKON gaming stage packages — Steam, Gamescope, MangoHud
# Requires RPM Fusion nonfree for Steam.
# i686 packages are still present in Fedora 45 (withdrawal proposal rejected).

# Steam client — native RPM from RPM Fusion nonfree (D-017)
steam

# Gamescope — Wayland nested compositor for gaming sessions
gamescope

# MangoHud — in-game performance overlay
mangohud
goverlay

# Controller and joystick support
gamemode
usbutils
steam-devices

# ── Vulkan (64-bit) ───────────────────────────────────────────────────────────
# CS2 and modern titles require Vulkan. mesa-vulkan-drivers provides the ICD
# for AMD (RADV) and Intel (ANV). nvidia variant adds its own ICD.
vulkan-loader
vulkan-tools
mesa-vulkan-drivers

# ── Vulkan + OpenGL (32-bit) — required for Steam itself and 32-bit titles ───
# Steam's own UI is a 32-bit process; without these it shows a blank window
# or refuses to open entirely.
vulkan-loader.i686
mesa-vulkan-drivers.i686
mesa-dri-drivers.i686
mesa-libGL.i686
mesa-libEGL.i686
mesa-libgbm.i686

# ── Other 32-bit runtime libs Steam needs ─────────────────────────────────────
libX11.i686
libXext.i686
libXrandr.i686
libXinerama.i686
libXi.i686
libxcb.i686
alsa-lib.i686
libpulse.i686
nss.i686
gtk2.i686
libvdpau.i686

# ── Proton / game compat ──────────────────────────────────────────────────────
# protontricks helps manage per-game Proton prefixes
python3-vdf
# xterm is used by some older Proton launchers as a console fallback
xterm
# zenity provides GUI dialogs (some installers need it)
zenity

# ── Video acceleration (improves video playback in-game / game cutscenes) ─────
libva
libva-utils
mesa-va-drivers
mesa-va-drivers.i686

# WINE compatibility (optional; most users rely on Proton inside Steam)
# Not installed here; available as Flatpak at user request.
